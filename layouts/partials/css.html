Pass a string path, or a dict with "path" and optional "id" / "media". */ -}} {{- $path := . }}{{ $attrs := "" }} {{- if reflect.IsMap . }} {{- $path = .path }} {{- with .id }}{{ $attrs = printf `%s id="%s"` $attrs . }}{{ end }} {{- with .media }}{{ $attrs = printf `%s media="%s"` $attrs . }}{{ end }} {{- end }} {{- with resources.Get $path | minify | fingerprint }} {{- else }}{{ errorf "css partial: asset %q not found" $path }}{{ end -}}
